CREATE STREAM pageviews_stream
  WITH (KAFKA_TOPIC='pageviews', VALUE_FORMAT='AVRO');

SELECT * FROM pageviews_stream EMIT CHANGES;

CREATE TABLE users_table (id VARCHAR PRIMARY KEY)
  WITH (KAFKA_TOPIC='users', VALUE_FORMAT='AVRO');
  
CREATE STREAM user_pageviews
  AS SELECT users_table.id AS userid, pageid, regionid, gender
    FROM pageviews_stream
    LEFT JOIN users_table ON pageviews_stream.userid = users_table.id
EMIT CHANGES;

# NOVITA': nuovo protocollo di rebalance (KIP-848), group.protocol=consumer
# --- Nuovo protocollo consumer group (KIP-848), GA in Kafka 4.0 ---
# Classic: il consumer "leader" del gruppo calcola l'assegnazione; a ogni ingresso/uscita di un membro tutti
#   rilasciano le partizioni e rifanno JoinGroup/SyncGroup (rebalance "stop the world").
# Nuovo: il group coordinator sul broker calcola l'assegnazione, nessun consumer leader e nessuna barriera
#   globale; il rebalance e' incrementale (un consumer perde solo le partizioni che gli vengono tolte).
#   Un solo heartbeat verso il coordinatore, che risponde con l'assegnazione.
#   session.timeout / heartbeat e strategia (uniform o range) si configurano lato broker/gruppo
#   (group.consumer.*), non piu' lato client (partition.assignment.strategy non si usa).
#   Si attiva lato client con group.protocol=consumer (default: classic).
# ConsumerDemoNewProtocol: e' ConsumerDemo con in piu' group.protocol=consumer; si sottoscrive a first_topic
#   e stampa i messaggi. Dimostra che il protocollo si abilita con una sola proprieta' e che il gruppo e'
#   davvero sul nuovo protocollo (describe --state: ASSIGNMENT-STRATEGY uniform, tipo Consumer).
#   NON mostra da sola il vantaggio (nessun blocco durante il rebalance): testato un solo consumer.
#   Per vederlo: avviare due consumer sullo stesso group.id (uno con classic, uno con consumer) e confrontare
#   i log: nel classic compaiono "Revoke previously assigned partitions" e "Join group" per tutti i membri.
mvn clean package
java -cp target/kafka-basics-jar-with-dependencies.jar kafka.tutorial1.ConsumerDemoNewProtocol
sudo docker exec kafka1 /opt/kafka/bin/kafka-consumer-groups.sh --list --type consumer --state --bootstrap-server kafka1:19092
sudo docker exec kafka1 /opt/kafka/bin/kafka-consumer-groups.sh --describe --group my-kip848-application --state --bootstrap-server kafka1:19092
# (ASSIGNMENT-STRATEGY: uniform; con group.protocol=classic e' range/cooperative-sticky)

#   Caso 1: at-least-once (duplicati lato consumer)   Caso 2: producer idempotente   Caso 3: transazioni
sudo docker compose -f docker-compose-kraft-single.yml up -d
sleep 20   # attendere il broker: se un consumer stampa a raffica NotCoordinatorException, aspettare e rilanciare
cd kafka-basics && mvn clean package && cd ..
J=kafka-basics/target/kafka-basics-jar-with-dependencies.jar
K="sudo docker exec kafka /opt/kafka/bin"
for t in eo-input eo-output eo-tx eo-idempotent-true eo-idempotent-false; do $K/kafka-topics.sh --create --topic $t --bootstrap-server localhost:9092; done

# ---------- CASO 1: AT-LEAST-ONCE (elaborazione e commit dell'offset sono due passi separati)
printf 'm1\nm2\nm3\n' | sudo docker exec -i kafka /opt/kafka/bin/kafka-console-producer.sh --topic eo-input --bootstrap-server localhost:9092
java -cp $J kafka.exactlyonce.AtLeastOnceConsumerDemo crash    # elabora m1 m2 m3, "crash" prima del commit
java -cp $J kafka.exactlyonce.AtLeastOnceConsumerDemo crash    # DUPLICATI: rielabora m1 m2 m3
java -cp $J kafka.exactlyonce.AtLeastOnceConsumerDemo commit   # rielabora ancora una volta, poi committa
java -cp $J kafka.exactlyonce.AtLeastOnceConsumerDemo commit   # niente: l'offset e' avanzato
$K/kafka-consumer-groups.sh --describe --group eo-at-least-once --bootstrap-server localhost:9092
# Messaggio: i messaggi sono arrivati 3 volte al "sistema esterno" (DB, REST...). Serve un consumer idempotente.

# ---------- CASO 2: PRODUCER IDEMPOTENTE (niente duplicati da retry del producer)
java -cp $J kafka.exactlyonce.IdempotentProducerDemo true      # default da Kafka 3.0
java -cp $J kafka.exactlyonce.IdempotentProducerDemo false
# Senza guasti le due esecuzioni sono identiche (100 messaggi, uno al secondo). Per vedere l'effetto dell'idempotenza
# serve provocare un retry: mentre il producer e' in esecuzione, da un altro terminale
#   sudo docker pause kafka ; sleep 8 ; sudo docker unpause kafka
# (pause e' il modo affidabile: il broker riceve la richiesta, la scrive, ma la risposta arriva dopo il timeout di 5s
#  e va persa; con "docker stop/start" il batch spesso non viene scritto e il duplicato non si forma)
# Nel log del producer compare "FALLIMENTO + RETRY". Poi confrontare il numero di messaggi nei due topic:
$K/kafka-get-offsets.sh --topic eo-idempotent-true --bootstrap-server localhost:9092    # 100
$K/kafka-get-offsets.sh --topic eo-idempotent-false --bootstrap-server localhost:9092   # >100 se il retry ha duplicato
# Il log del broker mostra solo che i batch con idempotenza portano producerId e numero di sequenza (il broker usa quelli per scartare i duplicati)
for v in true false; do
  echo "== idempotence=$v"
  sudo docker exec kafka sh -c "/opt/kafka/bin/kafka-dump-log.sh --files /tmp/kraft-combined-logs/eo-idempotent-$v-0/00000000000000000000.log | grep -oE 'baseOffset: [0-9]+|baseSequence: -?[0-9]+|producerId: -?[0-9]+' | paste -sd' '"
done
# true  -> baseOffset: 0 baseSequence: 0 producerId: 1
# false -> baseOffset: 0 baseSequence: -1 producerId: -1
# (garanzia valida per partizione e per la vita del producer: se si riavvia ottiene un nuovo producerId)

# ---------- CASO 3: TRANSAZIONI (exactly-once dentro Kafka)
# 3a) una transazione committata (3 msg) e una abortita (3 msg) sullo stesso topic  (la lettura con ReadCommittedConsumerDemo fa parte di 3a)
java -cp $J kafka.exactlyonce.TransactionalProducerDemo
java -cp $J kafka.exactlyonce.ReadCommittedConsumerDemo eo-tx read_committed     # 3 messaggi: "committed 0..2"
java -cp $J kafka.exactlyonce.ReadCommittedConsumerDemo eo-tx read_uncommitted   # 6 messaggi: anche "aborted 0..2"
# Con la console (read_committed e' ESPLICITO, il default dei consumer e' read_uncommitted):
$K/kafka-console-consumer.sh --topic eo-tx --from-beginning --bootstrap-server localhost:9092 --isolation-level read_committed --timeout-ms 15000
# NOTA: nel producer prima di abortTransaction() serve flush(), altrimenti i record nel buffer vengono scartati
#       e non arrivano nemmeno al broker.

# 3b) consume-transform-produce exactly-once: input -> maiuscolo -> output, offset committati NELLA STESSA transazione
# (l'input m1 m2 m3 e' gia' su eo-input: l'ha messo la printf del CASO 1; questa demo non produce input, lo rilegge con un consumer group suo)
java -cp $J kafka.exactlyonce.ExactlyOnceProcessorDemo abort    # scrive M1 M2 M3 su eo-output ma annulla: offset non avanza
java -cp $J kafka.exactlyonce.ExactlyOnceProcessorDemo commit   # rielabora e committa output + offset insieme
java -cp $J kafka.exactlyonce.ExactlyOnceProcessorDemo commit   # niente da fare: offset gia' a 3
java -cp $J kafka.exactlyonce.ReadCommittedConsumerDemo eo-output read_committed     # M1 M2 M3 UNA volta sola
java -cp $J kafka.exactlyonce.ReadCommittedConsumerDemo eo-output read_uncommitted   # 6 messaggi: i 3 annullati + i 3 validi
$K/kafka-consumer-groups.sh --describe --group eo-processor --bootstrap-server localhost:9092   # CURRENT-OFFSET 3, LAG 0
# Kafka Streams fa tutto questo con processing.guarantee=exactly_once_v2

# ---------- LIMITE: l'exactly-once vale SOLO dentro Kafka (topic -> topic).
# Se il consumer scrive su un DB / chiama un'API, Kafka non annulla l'effetto: rendere l'operazione idempotente
# (upsert con chiave univoca = id del messaggio) o salvare l'offset nella stessa transazione del DB.

sudo docker compose -f docker-compose-kraft-single.yml down -v
